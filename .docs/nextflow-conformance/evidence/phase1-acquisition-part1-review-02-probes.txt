parser = argparse.ArgumentParser()

SCRATCH = args.scratch.resolve()
SCRATCH.mkdir(parents=True, exist_ok=True)
RUN = Path(tempfile.mkdtemp(prefix='nextflow-probes-', dir=SCRATCH))

INPUTS = REPO / '.tmp/nextflow-conformance/test-inputs'
COMMIT = '232b60569865e9a4577e48c1955409238359d6ca'

sha = lambda b: hashlib.sha256(b).hexdigest()

def write_json(p, value):
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(value, indent=2) + '\n')

def digest_file(p):
    with p.open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()

def snapshot(root):
    rows = []
    for p in sorted(root.rglob('*')):
        s = p.lstat()
        row = {'path': str(p.relative_to(root)), 'mode': s.st_mode}
        if stat.S_ISREG(s.st_mode):
            row.update(bytes=s.st_size, sha256=digest_file(p))
        elif stat.S_ISLNK(s.st_mode):
            row['link'] = os.readlink(p)
        rows.append(row)
    return rows

source = (INPUTS / 'nextflow-source.tar.gz').read_bytes()
dist = (INPUTS / 'nextflow-26.04.6-dist').read_bytes()
tree = json.loads((INPUTS / 'nextflow-tree.json').read_text())
assert sha(source) == 'f1c889c683094cbba5eda8fab701dac48624253f39c41a8e5361729f14c6ba11'
assert len(dist) == 42355106 and sha(dist) == '182a63c74074e2dc7956ffa3c8cd59de952ed2c44394e21faf5e1736b945444c'
assert tree['sha'] == TREE and not tree['truncated']
contents = {}
with tarfile.open(fileobj=io.BytesIO(source), mode='r:gz') as archive:
    for member in archive:
        if member.isdir():
            continue
        name = member.name.split('/', 1)[1]
        assert name not in contents
        contents[name] = (member.linkname.encode() if member.issym()
                          else archive.extractfile(member).read())
files = []
for entry in tree['tree']:
    if entry['type'] != 'blob':
        continue
    content = contents[entry['path']]
    oid = hashlib.sha1(b'blob ' + str(len(content)).encode() + b'\x00' + content).hexdigest()
    assert oid == entry['sha']
    files.append({'path': entry['path'], 'blob': oid, 'mode': entry['mode'],
                  'sha256': sha(content), 'bytes': len(content),
                  'state': 'selected' if entry['path'] == SELECTED else 'unreviewed',
                  'rationale': None, 'review_id': None})
files.sort(key=lambda x:x['path'])
assert len(files) == len(contents) == 2856 and len(tree['tree']) == 3584
# Recompute every directory object independently, including the pinned root.
for parent, expected in [('', TREE)] + [(e['path'],e['sha']) for e in tree['tree'] if e['type']=='tree']:
    entries = [e for e in tree['tree'] if e['path'].rpartition('/')[0] == parent]
    entries.sort(key=lambda e:(e['path'].rsplit('/',1)[-1] + ('/' if e['type']=='tree' else '')).encode())
    raw = b''.join(e['mode'].lstrip('0').encode() + b' ' + e['path'].rsplit('/',1)[-1].encode()
                   + b'\0' + bytes.fromhex(e['sha']) for e in entries)
    actual = hashlib.sha1(b'tree ' + str(len(raw)).encode() + b'\0' + raw).hexdigest()
    assert actual == expected, parent

cert, key = RUN/'nextflow-fixture.pem', RUN/'nextflow-fixture.key'
cert_command = ['openssl','req','-x509','-newkey','rsa:2048','-nodes','-keyout',str(key),
                '-out',str(cert),'-days','1','-subj','/CN=localhost',
                '-addext','subjectAltName=IP:127.0.0.1']
with (RUN/'certificate.log').open('wb') as log:
    subprocess.run(cert_command, stdout=log, stderr=log, check=True)
state = {'source': source, 'requests': [], 'failure': False}
pom = b'<project><groupId>org.example</groupId><artifactId>nextflow-review</artifactId><version>1</version></project>\n'
class Handler(http.server.BaseHTTPRequestHandler):
    def do_GET(self):
        state['requests'].append(self.path)
        if state['failure']:
            self.send_response(503); self.end_headers(); return
        data = {'/SOURCE':state['source'], '/RUNTIME':dist, '/POM':pom}.get(self.path)
        if data is None:
            self.send_response(404); self.end_headers(); return
        self.send_response(200); self.send_header('Content-Length',str(len(data))); self.end_headers()
        self.wfile.write(data)
    def log_message(self, *_):
        pass
server = http.server.ThreadingHTTPServer(('127.0.0.1',0),Handler)
context = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
context.load_cert_chain(cert,key)
server.socket = context.wrap_socket(server.socket,server_side=True)
thread = threading.Thread(target=server.serve_forever,daemon=True)
thread.start()
origin = f'https://127.0.0.1:{server.server_port}'
env = os.environ | {'SSL_CERT_FILE':str(cert),'SSL_CERT_DIR':str(RUN/'no-extra-certs')}
results = []

def fixture(label):
    root, cache = RUN/label/'corpus', RUN/label/'cache'
    root.mkdir(parents=True); (cache/'java').mkdir(parents=True)
    java = b'Independent inert Java fixture; no Java execution\n'
    (cache/'java/release').write_bytes(java)
    objects = [('POM','dependency-metadata','file','nextflow-review.pom',pom),
               ('RUNTIME','runtime','opaque-dist','nextflow-26.04.6-dist',dist),
               ('SOURCE','source','extracted-archive','source.tar.gz',state['source'])]
    lock = {'schema':1, 'id':'NEXTFLOW_REVIEW_FIXTURE', 'origin':origin, 'revision':COMMIT,
            'tree':{'sha':TREE,'origin':origin+'/tree','truncated':False,'entries':[{k:e[k] for k in ('path','sha','mode','type')} for e in tree['tree']]},
            'files':files,'artifacts':[
                {'id':i,'role':r,'origin':origin+'/'+i,
                 'file':{'path':n,'sha256':sha(d),'bytes':len(d)},'packaging':p,
                 'coordinate':'org.example:nextflow-review:1' if i=='POM' else None,'dependencies':[]}
                for i,r,p,n,d in objects],
            'environment':{'java_home':'java','java_version':'inert independent fixture; never executed',
                'os':'linux','arch':'amd64','java_files':[{'file':{'path':'release','sha256':sha(java),'bytes':len(java)},
                'link':None,'mode':420}], 'variables':[],
                'dependencies':[{'id':'NEXTFLOW_REVIEW','group':'org.example','name':'nextflow-review',
                'version':'1','artifact_id':None,'pom_id':'POM'}]}}
    selected = next(f for f in files if f['path']==SELECTED)
    span = {'file':SELECTED,'start':0,'end':selected['bytes'],'sha256':selected['sha256']}
    replacements = {'BLOCKS':'BLOCK','OBLIGATIONS':'FACET','REQUIREMENTS':'REQ','UATS':'UAT',
                    'REVIEWS':'REVIEW','DECISIONS':'DECISION','BINDINGS':'BINDING'}
    for p in (REPO/'nextflowconformance/testdata/records').glob('*.json'):
        if p.stem == 'sources.lock':
            continue
        raw = p.read_text()
        for old,new in replacements.items():
            raw = raw.replace('"'+old+'"','"'+new+'"')
        value = json.loads(raw)
        if p.stem=='target':
            value.update(tag=TAG,commit=COMMIT,source_tree=TREE)
        if p.stem=='blocks':
            value['span'] = span
        if p.stem=='reviews':
            value['source_spans']=[span]
            value['input_hashes']['source']=[{'path':SELECTED,'sha256':selected['sha256'],'bytes':selected['bytes']}]
        if p.stem=='batches':
            value['source_spans']=[span]
        destination = root/p.name
        if p.stem=='attempts':
            destination = root/'attempts'/(value['id']+'.json')
        elif p.stem!='target':
            value=[value]
        write_json(destination,value)
    write_json(root/'sources.lock.json',lock)
    return root,cache

def invoke(label,root,cache,command='acquire',candidate=None):
    args=[str(BINARY),command,'--root',str(root),'--cache',str(cache)]
    if candidate:
        args += ['--lock-candidate',str(candidate)]
    before=len(state['requests'])
    result=subprocess.run(args,env=env,capture_output=True,timeout=90,check=False)
    evidence=RUN/label
    evidence.mkdir(exist_ok=True)
    (evidence/'stdout.json').write_bytes(result.stdout)
    (evidence/'stderr.txt').write_bytes(result.stderr)
    (evidence/'exit.txt').write_text(str(result.returncode)+'\n')
    output=json.loads(result.stdout)
    row={'label':label,'argv':args,'exit':result.returncode,'complete':output['complete'],
         'diagnostics':[d['code'] for d in output['diagnostics']],
         'verified_artifacts':output['counts']['verified_artifacts'],
         'requests':state['requests'][before:]}
    results.append(row)
    return row


violations=[]
def expect(row, code, diagnostic=None):
    row['expected_exit']=code
    row['expected_diagnostics']=[] if diagnostic is None else [diagnostic]
    if row['exit']!=code or row['diagnostics']!=row['expected_diagnostics']:
        violations.append(row['label'])

try:
    root,cache=fixture('boundary-base')
    row=invoke('boundary-base-acquire',root,cache)
    expect(row,0)
    assert row['exit']==0
    shutil.copyfile(cache/'sources.lock.candidate.json',root/'sources.lock.json')
    acquired=json.loads((root/'sources.lock.json').read_text())
    source_art=next(a for a in acquired['artifacts'] if a['role']=='source')
    source_tree=cache/(source_art['file']['path']+'-tree')
    pom_path=cache/next(a for a in acquired['artifacts'] if a['role']=='dependency-metadata')['file']['path']
    # Public caller can use directories or normalized relative paths as candidates.
    paths=[('cache-root',cache),('java-root',cache/'java'),
           ('source-tree-root',source_tree),('artifact-parent',pom_path.parent),
           ('normalized-java',cache/'unused/../java/release')]
    for label,candidate in paths:
        before=snapshot(cache); corpus_before=snapshot(root)
        row=invoke('ancestor-'+label,root,cache,candidate=candidate)
        expect(row,2,'E_SOURCE_PATH')
        row['cache_preserved']=snapshot(cache)==before
        row['corpus_preserved']=snapshot(root)==corpus_before
        assert row['cache_preserved'] and row['corpus_preserved']
    # Renaming a hardlinked candidate safely replaces its directory entry, not input bytes.
    hardlink=cache/'hardlinked-candidate.json'
    os.link(pom_path,hardlink)
    old_hash=digest_file(pom_path)
    row=invoke('hardlink-candidate',root,cache,candidate=hardlink)
    expect(row,0)
    row['input_preserved']=digest_file(pom_path)==old_hash
    assert row['input_preserved']
    shutil.copyfile(hardlink,root/'sources.lock.json')
    row=invoke('hardlink-candidate-validate',root,cache,'validate')
    expect(row,0)
    # Existing symlink outputs are rejected before any requests or writes.
    for label,link,candidate in [('leaf',cache/'candidate-link',cache/'candidate-link'),
                                 ('parent',cache/'candidate-dir-link',cache/'candidate-dir-link/new.json')]:
        os.symlink(cache/'java',link)
        before=snapshot(cache)
        row=invoke('symlink-'+label,root,cache,candidate=candidate)
        expect(row,2,'E_SOURCE_PATH')
        row['cache_preserved']=snapshot(cache)==before
        assert row['cache_preserved'] and not row['requests']
    # Each mutation changes one root header. Every pinned file stays byte-identical.
    root_targets=[('dot','.',0),('dot-slash','./',0),('normalized-dot','safe/..',0),
                  ('normalized-child','safe/../child',0),('absolute','/nextflow-outside',2),
                  ('parent','../nextflow-outside',2)]
    original_binary=BINARY
    for label,target,expected in root_targets:
        changed=io.BytesIO(); modified=0
        with tarfile.open(fileobj=io.BytesIO(source),mode='r:gz') as old, tarfile.open(fileobj=changed,mode='w:gz') as new:
            for member in old:
                if member.isdir() and '/' not in member.name.rstrip('/'):
                    member.type=tarfile.SYMTYPE;member.linkname=target;member.size=0;modified+=1
                    new.addfile(member)
                else:
                    new.addfile(member,old.extractfile(member) if member.isfile() else None)
        assert modified==1
        state['source']=changed.getvalue()
        archive_path=RUN/('nextflow-root-'+label+'.tar.gz')
        archive_path.write_bytes(state['source'])
        measured={}
        with tarfile.open(fileobj=io.BytesIO(state['source']),mode='r:gz') as archive:
            for member in archive:
                if '/' not in member.name.rstrip('/') or member.isdir(): continue
                measured[member.name.split('/',1)[1]]=member.linkname.encode() if member.issym() else archive.extractfile(member).read()
        assert measured==contents
        for variant in (['current','original'] if expected==0 else ['current']):
            BINARY=original_binary if variant=='current' else args.original_binary.resolve()
            case_root,case_cache=fixture('root-'+label+'-'+variant)
            previous=b'previous candidate preserved on failure\n'
            (case_cache/'sources.lock.candidate.json').write_bytes(previous)
            before=snapshot(case_cache)
            row=invoke('root-'+label+'-'+variant,case_root,case_cache)
            row.update(target=target,pinned_blobs=len(measured),tree_entries=len(tree['tree']),archive=str(archive_path))
            expect(row,expected,None if expected==0 else 'E_SOURCE_PATH')
            if row['exit']!=0:
                row['cache_preserved']=snapshot(case_cache)==before
                assert row['cache_preserved']
            else:
                shutil.copyfile(case_cache/'sources.lock.candidate.json',case_root/'sources.lock.json')
                row=invoke('root-'+label+'-'+variant+'-validate',case_root,case_cache,'validate')
                expect(row,0)
    BINARY=original_binary
    server.shutdown();server.server_close();thread.join()
    row=invoke('boundary-offline-validate',root,cache,'validate')
    expect(row,0)
    assert not row['requests']
finally:
    server.shutdown();server.server_close()
    write_json(RUN/'results.json',results)
    write_json(RUN/'artifact-tree.json',snapshot(RUN))
    write_json(SCRATCH/'latest-probes.json',{'run':str(RUN),'results':results,'violations':violations})
    print(json.dumps({'run':str(RUN),'count':len(results),'violations':violations},indent=2))
raise SystemExit(1 if violations else 0)
